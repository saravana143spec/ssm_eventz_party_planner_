        function formatINR(val) {
            return '₹' + val.toLocaleString('en-IN');
        }

        function calculateEstimator() {
            let guestSurcharge = 0;
            if (guestCount > 50) {
                guestSurcharge = (guestCount - 50) * 30;
            }

            let addonTotal = 0;
            addonCheckboxes.forEach(box => {
                if (box.checked) {
                    addonTotal += parseInt(box.dataset.price);
                }
            });

            let finalTotal = currentBasePrice + guestSurcharge + addonTotal;

            summaryBaseTheme.textContent = currentBaseTheme;
            summaryBaseCost.textContent = formatINR(currentBasePrice);
            summaryGuestCost.textContent = formatINR(guestSurcharge);
            summaryAddonCost.textContent = formatINR(addonTotal);
            summaryTotalCost.textContent = formatINR(finalTotal);

            formBudget.value = `${formatINR(finalTotal)} (${currentBaseTheme}, ${guestCount} Guests)`;
        }

                eventTypeBtns.forEach(b => {

                    b.classList.add('border-brand-cardBorder', 'bg-brand-dark/50');
                    b.querySelector('i').className = b.querySelector('i').className.replace('text-brand-gold', 'text-slate-400');
                });

                btn.classList.add('active', 'border-brand-gold', 'bg-brand-gold/10');
                btn.classList.remove('border-brand-cardBorder', 'bg-brand-dark/50');
                btn.querySelector('i').classList.add('text-brand-gold');

                currentBasePrice = parseInt(btn.dataset.price);
                currentBaseTheme = btn.querySelector('span').textContent;
                calculateEstimator();
            });

        // --- 5. WhatsApp Booking Submission Builder ---
        function handleWhatsAppBooking(e) {
            e.preventDefault();

            const name = document.getElementById('clientName').value.trim();
            const phone = document.getElementById('clientPhone').value.trim();
            const date = document.getElementById('eventDate').value;
            const venue = document.getElementById('venueLocation').value.trim();
            const budget = document.getElementById('formBudget').value;
            const notes = document.getElementById('specialNotes').value.trim();

            if (!name || !phone || !date || !venue) {
                showToast("Please complete all required fields!");
                return;
            }

            const targetWhatsApp = "919566047449"; 

            let msg = `*NEW EVENT INQUIRY - SSM EVENTZ*\n\n`;
            msg += `*Name:* ${name}\n`;
            msg += `*Contact:* ${phone}\n`;
            msg += `*Event Date:* ${date}\n`;
            msg += `*Venue Location:* ${venue}\n`;
            msg += `*Calculated Budget:* ${budget}\n`;
            if (notes) {
                msg += `*Special Notes:* ${notes}\n`;
            }
            msg += `\nPlease confirm availability and share theme photos!`;

            const encodedMsg = encodeURIComponent(msg);
            const waUrl = `https://wa.me/${targetWhatsApp}?text=${encodedMsg}`;

            window.open(waUrl, '_blank');
        }
